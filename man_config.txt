# Criar arquivos de script para ativar ambiente virtual e instalar requirements

# Windows batch script

## Instalar manualmente (copia/cola)

windows_script = """@echo off
REM Script para configurar ambiente virtual e instalar requirements
python -m venv venv
call venv\\Scripts\\activate
pip install -r requirements.txt
echo Ambiente configurado com sucesso!
pause
"""

# Linux/macOS shell script
unix_script = """#!/bin/bash
# Script para configurar ambiente virtual e instalar requirements
python3 -m venv venv
source venv/bin/activate
pip install -r requirements.txt
echo "Ambiente configurado com sucesso!"
"""
